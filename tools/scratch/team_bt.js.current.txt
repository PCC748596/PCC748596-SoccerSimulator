   820	    const profundidade = CAMPO_COMP * B.profundidade[compacLength];
   821	
   822	    /* --- centro em Z (Regra 1 e Regra 4) -------------------------------
   823	       CENTRO 5 METROS À FRENTE DA LINHA DA BOLA, com o sinal a depender da fase
   824	       No referencial de ataque da equipa (bb.dir):
   825	       - + é em direção ao ataque (à frente)
   826	       - - é em direção à defesa (atrás)
   827	    */
   828	    const dtMatch = (typeof Match !== 'undefined' && Match.delta) ? Match.delta : 0.016;
   829	    const reposta = (typeof Match !== 'undefined' && Match.state !== 'PLAY');
   830	
   831	    /*
   832	    NOVO SISTEMA DE BLOCOS (Defesa, Meio, Ataque)
   833	    As posições baseiam-se na linha da bola, mas têm comportamentos diferentes
   834	    dependendo da fase do jogo, evitando que o time estique num único retângulo gigante.
   835	    */
   836	    const targetOffsetZ = bb.isAttacking ? 5.0 : -5.0;
   837	
   838	    if (bb.blocoZSuave === undefined) {
   839	        bb.blocoZSuave = targetOffsetZ;
   840	    } else {
   841	        bb.blocoZSuave = seguirBola(bb.blocoZSuave, targetOffsetZ, BlockShape.seguimentoBola, dtMatch, reposta);
   842	    }
   843	
   844	    const bolaZDir = bb.bolaZSuave * bb.dir;
   845	    
   846	    // Configurações táticas do painel
   847	    const mentalBloco = (typeof MentalidadeModel !== 'undefined' &&
   848	        MentalidadeModel[Tatics.estilo] &&
   849	        typeof MentalidadeModel[Tatics.estilo].blocoZ === 'number')
   850	        ? MentalidadeModel[Tatics.estilo].blocoZ : 0;
   851	        
   852	    const capBase = (typeof TeamShape !== 'undefined' && TeamShape.linhaDefensiva) 
   853	        ? (TeamShape.linhaDefensiva[Tatics.linhaDefensiva] ?? TeamShape.linhaDefensiva.medium) 
   854	        : -15;
   855	        
   856	    const pesoMental = (typeof MentalidadeModel !== 'undefined' && typeof MentalidadeModel.pesoNaLinha === 'number')
   857	        ? MentalidadeModel.pesoNaLinha : 0;
   858	    const capLinha = capBase + mentalBloco * pesoMental;
   859	
   860	    let zDefBase;
   861	
   862	    if (bb.state === 'Offensive') {
   863	        // Time tem a bola e está construindo/atacando
   864	        zDefBase = bolaZDir - 15;
   865	    } else if (bb.state === 'T. Offensive') {
   866	        // Acabou de recuperar a bola (contra-ataque ou saída rápida)
   867	        zDefBase = bolaZDir - 10;
   868	    } else if (bb.state === 'T. Defensive') {
   869	        // Acabou de perder a bola
   870	        zDefBase = Math.min(bolaZDir - 20, capLinha); 
   871	    } else { 
   872	        // Defensive - Defendendo no próprio campo
   873	        zDefBase = Math.min(bolaZDir - 20, capLinha);
   874	    }
   875	    
   876	    // Se o time estiver com posse de bola, a defesa pode acompanhar lá de trás 
   877	    // mas nunca passar da linha da bola - 5m
   878	    if (bb.isAttacking) {
   879	        if (zDefBase > bolaZDir - 5) zDefBase = bolaZDir - 5;
   880	        
   881	        // Exceção: saída de bola com goleiro
   882	        let gkComBola = bb.ballCarrier && bb.ballCarrier.role === 'gk' && bb.ballCarrier.team === bb.team;
   883	        if (gkComBola) {
   884	            zDefBase = bolaZDir + 25;
   885	        }
   886	    }
   887	
   888	    // Incorpora adversário mais recuado e piso do GR na fase defensiva para o zDefBase
   889	    if (!bb.isAttacking) {
   890	        let maisRecuadoDir = null, pisoDir = null;
   891	        if (bb.opp && bb.opp.length) {
   892	            for (const o of bb.opp) {
   893	                if (!o || o.role === 'gk' || !o.model) continue;
   894	                const zDir = o.model.position.z * bb.dir;
   895	                if (maisRecuadoDir === null || zDir < maisRecuadoDir) maisRecuadoDir = zDir;
   896	            }
   897	        }
   898	        if (bb.own && bb.own.length) {
   899	            const gk = bb.own.find(pl => pl && pl.role === 'gk' && pl.model);
   900	            if (gk) pisoDir = gk.model.position.z * bb.dir + 1.0;
   901	        }
   902	
   903	        const M = (typeof MarkingModel !== 'undefined') ? MarkingModel : null;
   904	        const distancia = (M && M.distanciaPorPressao)
   905	            ? (M.distanciaPorPressao[Tatics.pressaoDefensiva] ?? M.distanciaPorPressao.balanced)
   906	            : 3.0;
   907	
   908	        zDefBase = recuoDaUltimaLinha(zDefBase, maisRecuadoDir, distancia, pisoDir, capLinha);
   909	    }
   910	
   911	    // O bloco agora é perfeitamente rígido para não se deformar.
   912	    // Ele é dividido em três terços com base na `profundidade` total da equipe.
   913	    let zDef = zDefBase;
   914	    let zMid = zDef + profundidade / 2;
   915	    let zAtk = zDef + profundidade;
   916	
   917	    const fundo = typeof LINHA_FUNDO !== 'undefined' ? LINHA_FUNDO : CAMPO_COMP / 2;
   918	    const profArea = typeof Area !== 'undefined' ? Area.profundidade : 16.5;
   919	    
   920	    // Limites Físicos do Campo
   921	    const minZ = -(fundo - profArea);
   922	    const maxZ = (fundo - 1.5);
   923	
   924	    // Se o ataque passar do limite do campo, empurramos o bloco inteiro para trás
   925	    if (zAtk > maxZ) {
   926	        const shift = maxZ - zAtk;
   927	        zAtk += shift;
   928	        zMid += shift;
   929	        zDef += shift;
   930	    }
   931	    
   932	    // Se a defesa passar do limite mínimo (própria área), empurramos o bloco inteiro para frente
   933	    if (zDef < minZ) {
   934	        const shift = minZ - zDef;
   935	        zDef += shift;
   936	        zMid += shift;
   937	        zAtk += shift;
   938	    }
   939	
   940	    // Mantemos z0 e z1 para compatibilidade com o resto do código
   941	    let z0 = zDef;
   942	    let z1 = zAtk;
   943	
   944	    /* --- largura e centro em X (Regra 1 e Regra 2) ----------------------
   945	       ACOMPANHA AS COORDENADAS DA BOLA EM X E LIMITA NOS LIMITES DO CAMPO (-34 a +34)
   946	    */
   947	    const largura = CAMPO_LARG * B.amplitude[compac];
   948	    const meiaLarg = largura / 2;
   949	
   950	    const centroX = bb.bolaXSuave;
   951	    let x0 = centroX - meiaLarg;
   952	    let x1 = centroX + meiaLarg;
   953	
   954	    const maxX = CAMPO_LARG / 2;
   955	    if (x0 < -maxX) {
   956	        x0 = -maxX;
   957	        x1 = x0 + largura;
   958	        if (x1 > maxX) x1 = maxX;
   959	    } else if (x1 > maxX) {
   960	        x1 = maxX;
   961	        x0 = x1 - largura;
   962	        if (x0 < -maxX) x0 = -maxX;
   963	    }
   964	
   965	    bb.bloco = {
   966	        x0: x0,
   967	        x1: x1,
   968	        z0: z0,
   969	        z1: z1,
   970	        zDef: zDef,
   971	        zMid: zMid,
   972	        zAtk: zAtk,
   973	        modo: modo
   974	    };
   975	
   976	    bb.defLineDir = z0;
   977	    bb.defLineZ = z0 * bb.dir;
   978	
   979	    bb.blockBottom = z0;
   980	    bb.blockTop = z1;
   981	
   982	    return bb.bloco;
   983	}

   985	/*
   986	DESLOCAMENTO LATERAL POR CORREDOR — quanto cada jogador desliza conforme o
   987	corredor onde a bola esta.
   988	
   989	Estava inline no slotNoBloco, com tres numeros a mao. Fora daqui nao havia
   990	maneira de testar o que interessa, que nao e cada numero por si: e a
   991	DISTANCIA ENTRE ELES depois de todos deslizarem.
   992	
   993	    ballCorredor   -1 esquerda, 0 eixo, +1 direita (mundo)
   994	    pCorredor      o mesmo, para o alvo deste jogador
   995	    isLateral      LB/RB/LM/RM/LWB/RWB/LW/RW
   996	    isDefesa       role 'def' — os centrais deslizam com a linha
   997	    isAttacking    a equipa tem a bola
   998	    sectorPedido   o painel pediu o corredor onde este jogador esta
   999	
  1000	Devolve os metros a somar ao x do alvo, ja com sinal do mundo.
  1001	
  1002	O que mudou, e porque (pedido explicito depois de ver em campo):
  1003	
  1004	    lado oposto, lateral   8.0 -> 5.0   fechava demais, ficava no eixo
  1005	    lado oposto, central   3.0 -> 4.0   passa a acompanhar o lateral
  1006	
  1007	O par 8/3 encolhia a distancia entre o lateral e o central do lado oposto em
  1008	5 m de uma vez, e os dois acabavam em cima um do outro. Fechar tem de ser um
  1009	movimento do bloco, nao de um homem so: agora sao 5 e 4, e a distancia entre
  1010	eles so encolhe 1 m.
  1011	
  1012	    mesmo lado, lateral    6.0 -> 4.0   abriam demais
  1013	    corredor central       2.0 -> 3.5 (defesas), 2.0 (o resto)
  1014	
  1015	Pura: sem Match, sem Tatics, sem THREE.
  1016	*/
  1017	function deslocamentoDeCorredor(o) {
  1018	    const ballCorredor = o.ballCorredor;
  1019	    const pCorredor = o.pCorredor;
  1020	
  1021	    if (ballCorredor === 0) {
  1022	        /*
  1023	        Bola no eixo: os corredores fechavam 4 m para dentro. Era o mais
  1024	        perverso dos estreitamentos — quanto MAIS central estava a bola, mais
  1025	        a equipa fechava, o oposto de jogar pelas pontas.
  1026	
  1027	        Quem esta num corredor que o painel pediu nao e puxado: se o senhor
  1028	        ligou a ala, e para haver alguem na ala quando a bola esta no meio,
  1029	        que e exactamente quando a ala serve para alguma coisa.
  1030	        */
  1031	        if (o.sectorPedido) return 0;
  1032	        return -pCorredor * 4.0;
  1033	    }
  1034	
  1035	    /*
  1036	    A BASCULAÇÃO JÁ FOI FEITA PELO BLOCO.
  1037	
  1038	    Estes deslocamentos — 3.5 m para o defesa no corredor central, 5.0 m para o
  1039	    lateral do lado oposto — foram calibrados quando o centro do rectângulo
  1040	    seguia a bola só a 0.7 em X (`BlockShape.basculacao`) e a equipa tinha de
  1041	    compensar por jogador. Com o bloco a acompanhar a bola 1:1, isto passou a
  1042	    somar-se por cima: medido, o lateral do lado oposto colava-se ao central
  1043	    (1.5 m) e os quatro defesas ficavam com os alvos a menos de 3 m uns dos
  1044	    outros em 92% do tempo.
  1045	
  1046	    Fica um resto pequeno, só para a linha não ficar perfeitamente recta: quem
  1047	    está do lado contrário ao da bola fecha um pouco, e mais nada.
  1048	    */
  1049	    if (pCorredor === 0) {
  1050	        return ballCorredor * (o.isDefesa ? 1.5 : 1.0);
  1051	    }
  1052	
  1053	    if (pCorredor === -ballCorredor) {
  1054	        return ballCorredor * (o.isLateral ? 1.0 : 1.25);
  1055	    }
  1056	
  1057	    // Mesmo lado da bola: o lateral abre para dar linha de passe pela ala, e
  1058	    // so com bola — sem ela, abrir e deixar o corredor interior a descoberto.
  1059	    if (o.isLateral && o.isAttacking) return ballCorredor * 4.0;
  1060	    return 0;
  1061	}

  1063	/*
  1064	Calcula a posição no mundo de um slot específico dentro do bloco tático.
  1065	*/
  1066	function calcularPontoDoSlot(slot, pos, role, fbStyle, bb) {
  1067	    const bloco = bb.bloco;
  1068	    if (!bloco || !slot) return null;
  1069	
  1070	    const linha = LineShape[role] || LineShape.mid;
  1071	    const desvioLinha = bb.isAttacking ? linha.comBola : linha.semBola;
  1072	
  1073	    let v = slot.v + desvioLinha;
  1074	
  1075	    // Ajuste fino por posicao especifica (lateral a frente do central, medio
  1076	    // de ponta sobe mais na construcao) - ver PositionDepthNudge.
  1077	    const nudge = PositionDepthNudge[pos];
  1078	    if (nudge) {
  1079	        if (bb.isAttacking) {
  1080	            let fbSt = (pos === 'LB' || pos === 'RB') ? FullBackStyle[fbStyle] : null;
  1081	            const isDefMental = (typeof Tatics !== 'undefined' && (Tatics.estilo === 'defesa' || Tatics.estilo === 'muito_defensiva'));
  1082	            let mult = fbSt ? fbSt.comBolaMult : 1;
  1083	            if (isDefMental) {
  1084	                if (pos === 'LB' || pos === 'RB') mult = FullBackStyle.defensive.comBolaMult;
  1085	                if (pos === 'LM' || pos === 'RM') mult = 0.0;
  1086	            }
  1087	            v += nudge.comBola * mult;
  1088	        } else {
  1089	            v += nudge.semBola;
  1090	        }
  1091	    }
  1092	    v = THREE.MathUtils.clamp(v, 0, 1);
  1093	
  1094	    // u: fecha lateralmente em torno do eixo central do bloco (0.5).
  1095	    const fechoLinha = LineShape.fecho[role] || LineShape.fecho.mid;
  1096	    const fecho = bb.isAttacking ? fechoLinha.comBola : fechoLinha.semBola;
  1097	
  1098	    const fechoSec = (typeof fechoDoSector === 'function' && typeof Tatics !== 'undefined')
  1099	        ? fechoDoSector(Tatics.setores)
  1100	        : 1.0;
  1101	        
  1102	    let fechoAdicional = 1.0;
  1103	    if (!bb.isAttacking && (pos === 'CB' || pos === 'DM' || pos === 'CM')) {
  1104	        const fatorCentral = Math.max(0, 1.0 - Math.abs(bb.bolaXSuave) / 18.0);
  1105	        fechoAdicional = 1.0 - (0.40 * fatorCentral); // aperta até mais 40% para fechar o meio
  1106	    }
  1107	
  1108	    const u = THREE.MathUtils.clamp(0.5 + (slot.u - 0.5) * fecho * fechoSec * fechoAdicional, 0.02, 0.98);
  1109	
  1110	    const MARGEM_LINHA_SLOT = 1.5;
  1111	    const limX = CAMPO_LARG / 2 - MARGEM_LINHA_SLOT;
  1112	    const limZ = CAMPO_COMP / 2 - MARGEM_LINHA_SLOT;
  1113	
  1114	    const dentro = (v0, v1, lim) => {
  1115	        if (v1 > lim) return -(v1 - lim);      // empurra para dentro pela direita
  1116	        if (v0 < -lim) return (-lim - v0);     // e pela esquerda
  1117	        return 0;
  1118	    };
  1119	    const empurraX = dentro(bloco.x0, bloco.x1, limX);
  1120	    const empurraZ = dentro(bloco.z0, bloco.z1, limZ);
  1121	
  1122	    let xTarget = bloco.x0 + u * (bloco.x1 - bloco.x0) + empurraX;
  1123	    
  1124	    let targetZRaw;
  1125	    if (v < 0.5) {
  1126	        // Interpola entre Defesa (v=0) e Meio (v=0.5)
  1127	        targetZRaw = bloco.zDef + (v / 0.5) * (bloco.zMid - bloco.zDef);
  1128	    } else {
  1129	        // Interpola entre Meio (v=0.5) e Ataque (v=1.0)
  1130	        targetZRaw = bloco.zMid + ((v - 0.5) / 0.5) * (bloco.zAtk - bloco.zMid);
  1131	    }
  1132	    
  1133	    let zTarget = (targetZRaw + empurraZ) * bb.dir;
  1134	
  1135	    const ballX = bb.ballX || 0;
  1136	    const CORREDOR_LIMITE = 11.33; 
  1137	
  1138	    let ballCorredor = 0;
  1139	    if (ballX > CORREDOR_LIMITE) ballCorredor = 1;
  1140	    else if (ballX < -CORREDOR_LIMITE) ballCorredor = -1;
  1141	
  1142	    let pCorredor = 0;
  1143	    if (xTarget > CORREDOR_LIMITE) pCorredor = 1;
  1144	    else if (xTarget < -CORREDOR_LIMITE) pCorredor = -1;
  1145	
  1146	    const isLateral = ['LB', 'RB', 'LM', 'RM', 'LWB', 'RWB', 'LW', 'RW'].includes(pos);
  1147	
  1148	    const meuSector = (typeof Tatics !== 'undefined' && Tatics.sectorDeX)
  1149	        ? Tatics.sectorDeX(xTarget, bb.dir)
  1150	        : 'cen';
  1151	    const sectorPedido = (typeof Tatics !== 'undefined' && Tatics.setores)
  1152	        ? Tatics.setores.indexOf(meuSector) >= 0
  1153	        : true;
  1154	
  1155	    xTarget += deslocamentoDeCorredor({
  1156	        ballCorredor: ballCorredor,
  1157	        pCorredor: pCorredor,
  1158	        isLateral: isLateral,
  1159	        isDefesa: role === 'def',
  1160	        isAttacking: bb.isAttacking,
  1161	        sectorPedido: sectorPedido
  1162	    });
  1163	
  1164	    xTarget = THREE.MathUtils.clamp(xTarget, bloco.x0 + empurraX, bloco.x1 + empurraX);
  1165	
  1166	    return {
  1167	        x: xTarget,
  1168	        z: zTarget
  1169	    };
  1170	}
